### 系统启动后频繁出现

```txt
istp-ish: timeout waiting for FW-initated reset
intel_ish_ipc 0000:00:12.0: ISH: hw start failed
rtw88_8821ce 0000:01:00.0: unhandled firmware c2h interrupt
```

```bash
# 禁用该模块
echo "blacklist intel_ish_ipc" | sudo tee /etc/modprobe.d/blacklist-intel-ish.conf
# 安装 iio-sensor-proxy 以启用传感器支持（如自动旋转屏幕等功能）
sudo pacman -S iio-sensor-proxy
# 启动并设置开机自启 iio-sensor-proxy 服务
sudo systemctl enable --now iio-sensor-proxy

# 解决输出和实际朝向相反的问题
# 获取当前设备的 DMI 信息
cat /sys/class/dmi/id/modalias
# 创建修正配置文件
sudo vim /etc/udev/hwdb.d/61-sensor-local.hwdb
# sensor:modalias:dmi:bvnAmericanMegatrendsInternational,LLC.:bvrN1305.23072601.T60.12W.12-16.VC3.006:bd11/21/2023:br5.25:efr1.1:svnDERE:pnDefaultstring:pvrDefaultstring:rvnNYM:rnN1305:rvrDefaultstring:cvnDefaultstring:ct10:cvrDefaultstring:skuDefaultstring:*
#  ACCEL_MOUNT_MATRIX=-1, 0, 0; 0, -1, 0; 0, 0, 1
# 保存并退出
# 更新 hwdb 数据库
sudo systemd-hwdb update
# 重新触发 udev 规则应用新的配置
sudo udevadm trigger -v -p /sys/class/iio/iio:device*

# 查看传感器输出
monitor-sensor


# 解决方案 1: 关闭网卡电源节能（优先尝试）
vim /etc/modprobe.d/rtw88.conf
# 写入以下内容:
# options rtw88_pci disable_aspm=1
# options rtw88_core disable_lps_deep=y
# 保存并重启

# 解决方案 2：更换为 AUR 的专有驱动（彻底解决）
# Linux 内核自带的 rtw88 驱动对 8821CE 的支持在某些机型上非常糟糕
# 社区维护的 rtl8821ce 专有驱动通常更稳定
vim /etc/modprobe.d/blacklist-rtw88.conf
# blacklist rtw88_8821ce
# blacklist rtw88_pci
# blacklist rtw88_core
# 屏蔽内核自带的 rtw88 驱动

# 安装 AUR 驱动 (需要先安装 base-devel 和 linux-headers)
pacman -S linux-headers base-devel
# 安装 rtl8821ce-dkms-git
yay -S rtl8821ce-dkms-git
# 重启

# 卸载 rtl8821ce-dkms-git
yay -Rns rtl8821ce-dkms-git
```

### 声卡初始化失败问题

内核模块已加载，但 SOF 固件未能正确初始化声卡，导致无声音输出

```bash
sudo vim /etc/default/grub
# 修改以下内容: 禁用 SOF，强制使用传统 HDA 驱动
# GRUB_CMDLINE_LINUX_DEFAULT="..."
# 在引号内添加 snd_intel_dspcfg.dsp_driver=1
# 保存并退出

# 重新生成 GRUB 配置文件
sudo grub-mkconfig -o /boot/grub/grub.cfg
```

## 通过以下步骤尝试启用 zram 内存压缩和交换空间，但是用 fastfetch 查看发现 Swap: Disabled 可能是什么原因？

```bash
# zram-generator -> 动态创建和管理 zram 设备的工具
pacman -S zram-generator
# 编辑配置文件
vim /etc/systemd/zram-generator.conf
# 添加以下内容:
[zram0]
zram-size = ram    # 使用全部物理内存大小作为 zram 交换空间
compression-algorithm = lz4

# 编辑 GRUB 源文件
vim /etc/default/grub
# 修改以下内容:
# GRUB_CMDLINE_LINUX_DEFAULT="loglevel=5 zswap.enabled=0"
# 禁用 zswap，避免与 zram 冲突
# 重新生成 GRUB 配置文件，使更改生效
grub-mkconfig -o /boot/grub/grub.cfg
```
